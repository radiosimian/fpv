#$ TITLE: 2-inch 2S Base (from Tango20)
#$ FIRMWARE_VERSION: 2026.6
#$ CATEGORY: TUNE
#$ STATUS: EXPERIMENTAL
#$ KEYWORDS: 2inch, 2S, micro, walksnail, rpm filter
#$ AUTHOR: radiosimian
#$ DESCRIPTION: Base configuration for 2S 2-inch builds, derived from Tango20 (GEPRC Taker F405 AIO).
#$ DESCRIPTION: Board-specific settings are NOT included: UART/serial assignments, PINIO, acc calibration,
#$ DESCRIPTION: vbat/ibata scale, board alignment and craft name. Set those per build.
#$ DESCRIPTION: Apply rates and battery settings with the target rate/battery profile selected.

# ---------- Core: always applied ----------

# Features and beeper
feature TELEMETRY
feature LED_STRIP
feature OSD
beeper -ON_USB
beacon RX_LOST

# Allow arming at any angle (needed for turtle mode after a crash)
set small_angle = 180

# Blackbox
set blackbox_sample_rate = 1/2
set blackbox_disable_bat = ON
set blackbox_disable_mag = ON
set blackbox_disable_alt = ON
set blackbox_disable_rssi = ON
set blackbox_disable_servos = ON
set blackbox_disable_gps = ON

# Rates (Actual: 100 deg/s centre, 800 deg/s max, 0.50 expo)
set roll_rc_rate = 10
set pitch_rc_rate = 10
set yaw_rc_rate = 10
set roll_expo = 50
set pitch_expo = 50
set yaw_expo = 50
set roll_srate = 80
set pitch_srate = 80
set yaw_srate = 80

# Pilot
set pilot_name = radiosimian

#$ OPTION BEGIN (CHECKED): Motors, bidirectional DShot and RPM-filter tune (12-pole motors)
set dshot_bidir = ON
set motor_pwm_protocol = DSHOT300
set motor_poles = 12
set pid_process_denom = 2
# RPM filter handles motor noise, so gyro LPF1 is off and a single dynamic notch is kept
set gyro_lpf1_static_hz = 0
set gyro_lpf1_dyn_min_hz = 0
set gyro_lpf1_dyn_max_hz = 0
set dyn_notch_count = 1
set dyn_notch_q = 500
#$ OPTION END

#$ OPTION BEGIN (CHECKED): LiHV batteries (4.35V max cell)
set vbat_max_cell_voltage = 435
#$ OPTION END

#$ OPTION BEGIN (CHECKED): HD OSD layout (Walksnail) and post-flight stats
set osd_rssi_dbm_alarm = -96
set osd_vbat_pos = 2563
set osd_link_quality_pos = 7660
set osd_rssi_dbm_pos = 7692
set osd_tim_2_pos = 3628
set osd_flymode_pos = 2147
set osd_throttle_pos = 2584
set osd_current_pos = 2614
set osd_mah_drawn_pos = 2595
set osd_craft_name_pos = 2083
set osd_altitude_pos = 6211
set osd_avg_cell_voltage_pos = 6627
set osd_disarmed_pos = 6485
set osd_log_status_pos = 3247
set osd_stat_bitmask = 8476652
#$ OPTION END

#$ OPTION BEGIN (UNCHECKED): Modes and HUD-profile adjustment (same radio/switch layout)
# AUX1 high: ARM
aux 0 0 0 1800 2100 0 0
# AUX3 middle: ANGLE (AUX3 is a 3-pos switch: low = acro, mid = angle, high = turtle)
aux 1 1 2 1400 1600 0 0
# AUX4 high: BEEPER
aux 2 13 3 1800 2100 0 0
# AUX7 low: BLACKBOX
aux 3 26 6 900 1200 0 0
# AUX8 high: BLACKBOX ERASE
aux 4 31 7 1800 2100 0 0
# AUX3 high: FLIP OVER AFTER CRASH (turtle)
aux 5 35 2 1800 2100 0 0
# AUX2 full range: HUD (OSD) profile select
adjrange 0 0 1 900 2100 29 1 0 0
#$ OPTION END
